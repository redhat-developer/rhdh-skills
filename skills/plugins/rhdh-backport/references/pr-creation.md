# PR Creation Patterns

Standard patterns for creating PRs during the backport workflow.

## PR #1: Backport Branch → Release Branch

```bash
create_pr1_backport_to_release() {
  local PR_NUM=$1
  local PLUGIN=$2
  local RELEASE=$3
  local BACKPORT_BRANCH=$4
  local RELEASE_BRANCH="${PLUGIN}/release-${RELEASE}"
  local PR_TITLE=$5
  local PR_URL=$6
  
  echo "📝 Creating PR #1: $BACKPORT_BRANCH → $RELEASE_BRANCH"
  
  # Get fork owner
  FORK_OWNER=$(gh api user --jq .login)
  
  # Create PR
  gh pr create \
    --repo redhat-developer/rhdh-plugins \
    --base "$RELEASE_BRANCH" \
    --head "${FORK_OWNER}:${BACKPORT_BRANCH}" \
    --title "backport: PR #${PR_NUM} to release-${RELEASE}" \
    --body "Cherry-picked from #${PR_NUM} for ${RELEASE} release

## Original PR

${PR_URL}

## Details

- **Plugin:** ${PLUGIN}
- **Release:** ${RELEASE}
- **Original title:** ${PR_TITLE}

## Workflow

This is **part 1/2** of the automated backport workflow:
1. ✅ This PR: Merge changes into \`${RELEASE_BRANCH}\`
2. ⏳ Next: Auto-create PR from \`${RELEASE_BRANCH}\` → \`workspace/${PLUGIN}\`
3. ⏳ Then: Version Packages PR will be auto-generated

After merge, the automation will continue with step 2.

---
🤖 Auto-generated by backport-auto skill"
  
  PR1_NUM=$(gh pr view --json number --jq '.number')
  
  if [ -z "$PR1_NUM" ]; then
    echo "❌ Failed to create PR #1"
    return 1
  fi
  
  echo "✅ PR #1 created: #$PR1_NUM"
  echo "   ${FORK_OWNER}:${BACKPORT_BRANCH} → ${RELEASE_BRANCH}"
  
  export PR1_NUM
  
  return 0
}
```

## PR #2: Release Branch → Workspace

**CRITICAL:** Must be from `upstream`, NOT fork!

```bash
create_pr2_release_to_workspace() {
  local PR_NUM=$1
  local PR1_NUM=$2
  local PLUGIN=$3
  local RELEASE=$4
  local RELEASE_BRANCH="${PLUGIN}/release-${RELEASE}"
  local WORKSPACE_BRANCH="workspace/${PLUGIN}"
  local PR_URL=$5
  
  echo "📝 Creating PR #2: $RELEASE_BRANCH → $WORKSPACE_BRANCH"
  echo "   ⚠️  MUST be from upstream to trigger Version Packages"
  
  # Verify release branch has merged changes from PR #1
  git fetch upstream
  
  # Create PR from upstream branch
  gh pr create \
    --repo redhat-developer/rhdh-plugins \
    --base "$WORKSPACE_BRANCH" \
    --head "$RELEASE_BRANCH" \
    --title "chore: sync ${PLUGIN} release-${RELEASE} to workspace" \
    --body "Backport of #${PR_NUM} to release ${RELEASE}

## Backport Chain

- **Original PR:** ${PR_URL}
- **Backport PR #1:** #${PR1_NUM} (merged)
- **This PR:** Part 2/2 - Sync to workspace

## Purpose

This PR:
1. Brings backported changes from \`${RELEASE_BRANCH}\` into \`${WORKSPACE_BRANCH}\`
2. **Triggers the Version Packages workflow**
3. After merge, Version Packages PR will update version numbers and CHANGELOG

## Important

⚠️ **Do not edit this PR manually**

This PR is auto-generated and should be merged as-is after CI passes.
Any manual changes may interfere with Version Packages generation.

---
🤖 Auto-generated by backport-auto skill

**Workflow:** This PR was created from \`upstream:${RELEASE_BRANCH}\` (not a fork) to ensure Version Packages workflow triggers correctly."
  
  PR2_NUM=$(gh pr view --json number --jq '.number')
  
  if [ -z "$PR2_NUM" ]; then
    echo "❌ Failed to create PR #2"
    return 1
  fi
  
  # Verify it's from upstream
  PR2_HEAD=$(gh pr view $PR2_NUM --json headRefName --jq '.headRefName')
  PR2_HEAD_REPO=$(gh pr view $PR2_NUM --json headRepositoryOwner --jq '.headRepositoryOwner.login')
  
  if [ "$PR2_HEAD_REPO" != "redhat-developer" ]; then
    echo "❌ Error: PR #2 created from fork!"
    echo "   Head: ${PR2_HEAD_REPO}:${PR2_HEAD}"
    echo "   Expected: redhat-developer:${RELEASE_BRANCH}"
    echo ""
    echo "This will NOT trigger Version Packages workflow"
    echo "Close this PR and investigate"
    return 1
  fi
  
  echo "✅ PR #2 created: #$PR2_NUM"
  echo "   redhat-developer:${RELEASE_BRANCH} → ${WORKSPACE_BRANCH}"
  echo "   ✅ Confirmed: From upstream (will trigger Version Packages)"
  
  export PR2_NUM
  
  return 0
}
```

## Changelog PR: Branch → Main

```bash
create_changelog_pr() {
  local PLUGIN=$1
  local RELEASE=$2
  local PR_NUM=$3
  local VP_VERSION=$4
  local VP_PR_NUM=$5
  local PR_TITLE=$6
  
  echo "📝 Creating changelog PR to main"
  
  # Get fork owner
  FORK_OWNER=$(gh api user --jq .login)
  CHANGELOG_BRANCH="changelog/${PLUGIN}-${RELEASE}-pr${PR_NUM}"
  
  # Create PR
  gh pr create \
    --repo redhat-developer/rhdh-plugins \
    --base main \
    --head "${FORK_OWNER}:${CHANGELOG_BRANCH}" \
    --title "docs: add ${PLUGIN} ${RELEASE} changelog for backport #${PR_NUM}" \
    --body "Adds changelog entry for backported PR #${PR_NUM}

## Backport Details

- **Original PR:** #${PR_NUM} - ${PR_TITLE}
- **Release:** ${RELEASE}
- **Version:** ${VP_VERSION}
- **Plugin:** ${PLUGIN}
- **Version Packages PR:** #${VP_PR_NUM}

## Changes

- Updated \`CHANGELOG.md\` to document this backport

## Purpose

This PR tracks what was backported to the ${RELEASE} release branch.
It ensures the main branch CHANGELOG reflects all released changes.

---
🤖 Auto-generated by backport-auto skill"
  
  CHANGELOG_PR=$(gh pr view --json number --jq '.number')
  
  if [ -z "$CHANGELOG_PR" ]; then
    echo "❌ Failed to create changelog PR"
    return 1
  fi
  
  echo "✅ Changelog PR created: #$CHANGELOG_PR"
  echo "   ${FORK_OWNER}:${CHANGELOG_BRANCH} → main"
  
  export CHANGELOG_PR_NUM=$CHANGELOG_PR
  
  return 0
}
```

## PR Labels

Add labels to PRs for easy tracking:

```bash
# Add labels to backport PR #1
gh pr edit $PR1_NUM \
  --repo redhat-developer/rhdh-plugins \
  --add-label "backport" \
  --add-label "release-${RELEASE}" \
  --add-label "plugin:${PLUGIN}"

# Add labels to sync PR #2
gh pr edit $PR2_NUM \
  --repo redhat-developer/rhdh-plugins \
  --add-label "backport" \
  --add-label "release-${RELEASE}" \
  --add-label "automated"

# Add labels to changelog PR
gh pr edit $CHANGELOG_PR_NUM \
  --repo redhat-developer/rhdh-plugins \
  --add-label "documentation" \
  --add-label "changelog"
```

## Link PRs Together

Add comments linking the PR chain:

```bash
# On original PR
gh pr comment $PR_NUM \
  --repo redhat-developer/rhdh-plugins \
  --body "✅ Backported to ${RELEASE} in #${PR1_NUM}"

# On PR #1
gh pr comment $PR1_NUM \
  --repo redhat-developer/rhdh-plugins \
  --body "🔗 Part of backport chain:
  - Original: #${PR_NUM}
  - Step 2: #${PR2_NUM} (workspace sync)"

# On PR #2
gh pr comment $PR2_NUM \
  --repo redhat-developer/rhdh-plugins \
  --body "🔗 Part of backport chain:
  - Original: #${PR_NUM}
  - Step 1: #${PR1_NUM} (merged)
  - Version Packages: Will be auto-created"
```

## Error Handling

**PR creation fails:**
```
❌ Failed to create PR

Possible reasons:
  1. No changes to create PR
  2. PR already exists
  3. Base branch doesn't exist
  4. Permission denied

Check: gh pr list --base {base_branch}
```

**Wrong head repository:**
```
❌ Error: PR #2 created from fork!
Head: username:release-1.10
Expected: redhat-developer:release-1.10

This will NOT trigger Version Packages workflow
Close this PR and investigate
```

**No permission:**
```
❌ Error: Resource not accessible by personal access token

You need write access to create PR
Request access or use gh auth login with correct scopes
```
